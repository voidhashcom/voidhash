---
title: Calendar
description: A date picker component for selecting dates.
---

## Usage

<Preview>
  <Calendar mode="single" className="rounded-md border" />
</Preview>

```tsx
import { Calendar } from "@voidhash/ui";

<Calendar mode="single" />;
```

## Examples

### Single Date Selection

<Preview>
  <Calendar mode="single" className="rounded-md border" />
</Preview>

```tsx
const [date, setDate] = useState<Date | undefined>(new Date());

<Calendar mode="single" selected={date} onSelect={setDate} />;
```

### Range Selection

<Preview>
  <Calendar mode="range" numberOfMonths={2} className="rounded-md border" />
</Preview>

```tsx
<Calendar mode="range" numberOfMonths={2} selected={dateRange} onSelect={setDateRange} />
```

### Multiple Months

<Preview>
  <Calendar mode="single" numberOfMonths={2} className="rounded-md border" />
</Preview>

```tsx
<Calendar mode="single" numberOfMonths={2} />
```

### Disabled Dates

```tsx
<Calendar mode="single" disabled={(date) => date < new Date()} />
```
