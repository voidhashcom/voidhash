---
title: Label
description: Renders an accessible label associated with form controls.
---

## Usage

<Preview>
  <div className="grid gap-2">
    <Label htmlFor="name">Name</Label>
    <Input id="name" placeholder="Enter your name" />
  </div>
</Preview>

```tsx
import { Label } from "@voidhash/ui";
import { Input } from "@voidhash/ui";

<div className="grid gap-2">
  <Label htmlFor="name">Name</Label>
  <Input id="name" placeholder="Enter your name" />
</div>;
```

## Examples

### With Checkbox

<Preview>
  <div className="flex items-center gap-2">
    <Checkbox id="agree" />
    <Label htmlFor="agree">I agree to the terms</Label>
  </div>
</Preview>

```tsx
<div className="flex items-center gap-2">
  <Checkbox id="agree" />
  <Label htmlFor="agree">I agree to the terms</Label>
</div>
```

### With Required Indicator

<Preview>
  <div className="grid gap-2">
    <Label htmlFor="required-field">
      Email <span className="text-destructive">*</span>
    </Label>
    <Input id="required-field" type="email" required />
  </div>
</Preview>

```tsx
<Label htmlFor="email">
  Email <span className="text-destructive">*</span>
</Label>
```

### Disabled State

The label automatically adjusts its style when associated with a disabled input.

<Preview>
  <div className="grid gap-2">
    <Label htmlFor="disabled-input">Disabled Field</Label>
    <Input id="disabled-input" disabled placeholder="Disabled" />
  </div>
</Preview>
