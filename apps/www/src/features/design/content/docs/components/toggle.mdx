---
title: Toggle
description: A two-state button that can be either on or off.
---

## Usage

<Preview>
  <Toggle aria-label="Toggle bold">
    <BoldIcon />
  </Toggle>
</Preview>

```tsx
import { Toggle } from "@voidhash/ui";
import { BoldIcon } from "lucide-react";

<Toggle aria-label="Toggle bold">
  <BoldIcon />
</Toggle>;
```
