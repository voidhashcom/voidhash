```tsx
import {
  Pagination,
  PaginationContent,
  PaginationEllipsis,
  PaginationItem,
  PaginationLink,
  PaginationNext,
  PaginationPrevious,
} from "@voidhash/ui";

<Pagination>
  <PaginationContent>
    <PaginationItem>
      <PaginationPrevious href="#" />
    </PaginationItem>
    <PaginationItem>
      <PaginationLink href="#">1</PaginationLink>
    </PaginationItem>
    <PaginationItem>
      <PaginationLink href="#" isActive>
        2
      </PaginationLink>
    </PaginationItem>
    <PaginationItem>
      <PaginationNext href="#" />
    </PaginationItem>
  </PaginationContent>
</Pagination>;
```

## Components

### Pagination

The root navigation container.

### PaginationContent

The list container for pagination items.

### PaginationItem

A single item in the pagination.

### PaginationLink

A page number link.

### PaginationPrevious

The previous page button.

### PaginationNext

The next page button.

### PaginationEllipsis

An ellipsis indicator for skipped pages.
